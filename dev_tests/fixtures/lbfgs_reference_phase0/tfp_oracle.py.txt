import numpy as np
import tensorflow as tf
import tensorflow_probability as tfp
from tensorflow_probability.python.optimizer import bfgs_utils

tf.config.threading.set_intra_op_parallelism_threads(1)
tf.config.threading.set_inter_op_parallelism_threads(1)
assert not tf.config.list_physical_devices('GPU')
assert not tf.config.list_logical_devices('GPU')

def scalar(value):
    value = value.item()
    if isinstance(value, float) and not np.isfinite(value):
        return 'nan' if np.isnan(value) else ('+inf' if value > 0 else '-inf')
    return value

def array(value):
    value = value.numpy()
    return {'shape': list(value.shape), 'values': [scalar(v) for v in value.reshape(-1)],
            'raw_hex': value.tobytes().hex(), 'dtype': str(value.dtype)}

def result_record(result):
    return {
        'position': array(result.position), 'objective_value': array(result.objective_value),
        'objective_gradient': array(result.objective_gradient),
        'converged': bool(result.converged.numpy()), 'failed': bool(result.failed.numpy()),
        'num_iterations': int(result.num_iterations.numpy()),
        'num_objective_evaluations': int(result.num_objective_evaluations.numpy()),
        'position_deltas': array(result.position_deltas),
        'gradient_deltas': array(result.gradient_deltas),
    }

records = []
for dtype in (tf.float32, tf.float64):
    def quadratic(scale, offset=0.):
        scale, offset = tf.constant(scale, dtype), tf.constant(offset, dtype)
        def objective(x):
            return offset + .5 * scale * tf.reduce_sum(x * x), scale * x
        return objective
    def flat_quartic(x):
        return tf.constant(2. ** 20, dtype) + .125 * tf.reduce_sum(x ** 4), .5 * x ** 3
    for name, objective in (
        ('unit_quadratic', quadratic(1.)),
        ('small_quadratic', quadratic(1e-8)),
        ('flat_offset_quadratic', quadratic(.125, 2. ** 20)),
        ('flat_offset_quartic', flat_quartic),
    ):
        initial = tf.constant([1.], dtype)
        initial_value, initial_gradient = objective(initial)
        r = tfp.optimizer.lbfgs_minimize(objective, initial_position=initial, **options)
        records.append({'case': name, 'dtype': dtype.name,
                        'initial_value': array(initial_value),
                        'initial_gradient': array(initial_gradient), 'result': result_record(r)})

    diagonal = tf.constant([1., 10.], dtype)
    def objective(x):
        return .5 * tf.reduce_sum(diagonal * x * x), diagonal * x
    first = tfp.optimizer.lbfgs_minimize(objective, initial_position=tf.constant([1., 1.], dtype),
        **dict(options, max_iterations=1))
    resumed = tfp.optimizer.lbfgs_minimize(objective, initial_position=None, previous_optimizer_results=first,
        **dict(options, max_iterations=2))
    fresh = tfp.optimizer.lbfgs_minimize(objective, initial_position=tf.constant([1., 1.], dtype),
        **dict(options, max_iterations=2))
    records.append({'case': 'cumulative_resume', 'dtype': dtype.name,
                    'first': result_record(first), 'resumed': result_record(resumed),
                    'fresh_two': result_record(fresh)})

    for name, offset in (('positive_relative_stop', 100.), ('negative_relative_continues', -100.)):
        def shifted(x):
            value, gradient = objective(x)
            return tf.constant(offset, dtype) + value, gradient
        r = tfp.optimizer.lbfgs_minimize(shifted, initial_position=tf.constant([1., 1.], dtype),
            **dict(options, max_iterations=1, tolerance=0., x_tolerance=0., f_relative_tolerance=.1))
        records.append({'case': name, 'dtype': dtype.name, 'result': result_record(r),
                        'overrides': {'max_iterations': 1, 'tolerance': 0., 'x_tolerance': 0., 'f_relative_tolerance': .1}})

    def unbounded(x):
        return tf.reduce_sum(x), tf.ones_like(x)
    failed = tfp.optimizer.lbfgs_minimize(unbounded, initial_position=tf.constant([0.], dtype), **options)
    records.append({'case': 'unbounded_linear', 'dtype': dtype.name, 'result': result_record(failed)})

    for name, value, gradient in (
        ('positive_infinity', float('inf'), 1.),
        ('negative_infinity', -float('inf'), 1.),
        ('nan_gradient', 1., float('nan')),
    ):
        def invalid(x):
            return tf.constant(value, dtype), tf.fill(tf.shape(x), tf.constant(gradient, dtype))
        r = tfp.optimizer.lbfgs_minimize(invalid, initial_position=tf.constant([0.], dtype), **options)
        records.append({'case': name, 'dtype': dtype.name, 'result': result_record(r)})

# Isolate signed-old-f relative tolerance and all four independent boundaries.
boundaries = []
for name, old_f, new_f, dx, g, gt, rt, xt, at in (
    ('relative_positive', 8., 9., 1., 1., 0., .125, 0., 0.),
    ('relative_negative', -8., -7., 1., 1., 0., .125, 0., 0.),
    ('absolute_equality', -8., -8., 1., 1., 0., 0., 0., 0.),
    ('gradient_boundary', 8., 9., 1., .125, .125, 0., 0., 0.),
    ('position_boundary', 8., 9., .125, 1., 0., 0., .125, 0.),
    ('absolute_boundary', -8., -7., 1., 1., 0., 0., 0., 1.),
):
    cast = lambda v: tf.constant(v, tf.float64)
    yes = bfgs_utils._check_convergence(cast([0.]), cast([dx]), cast(old_f), cast(new_f),
        cast([g]), cast(gt), cast(rt), cast(xt), cast(at))
    boundaries.append({'case': name, 'old_f': old_f, 'new_f': new_f, 'dx': dx, 'gradient': g,
        'gradient_tolerance': gt, 'relative_tolerance': rt, 'x_tolerance': xt,
        'absolute_tolerance': at, 'converged': bool(yes.numpy())})

    'resume_limits': [1, 2], 'resume_note': 'Analytic finite-limit test only; production budgets remain 200 fresh calls of max_iterations=1000.',
    'sources': sources, 'cases': records, 'convergence_boundaries': boundaries,
